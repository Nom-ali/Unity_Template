        private void DrawPackages()
        {
            TemplateSetupState state = TemplateSetupState.instance;
            selectedPackageTab = GUILayout.Toolbar(selectedPackageTab, new[] { "Unity Registry", "Unity Services", "Third Party" }, GUILayout.Height(30));
            PackageGroup group = (PackageGroup)selectedPackageTab;
            if (group == PackageGroup.UnityRegistry) DrawFeatureBundles();
            Box("Package selection", () => EditorGUILayout.LabelField("Ready is green. Selecting an uninstalled package checks only its latest version. Custom versions are available only for third-party and extra packages.", EditorStyles.wordWrappedMiniLabel));
            HashSet<string> featureDependencies = group == PackageGroup.UnityRegistry ? new HashSet<string>(FeatureBundles.SelectMany(f => f.Dependencies), StringComparer.OrdinalIgnoreCase) : new HashSet<string>();
            foreach (PackageDefinition package in PackageCatalog.Where(p => p.Group == group && !featureDependencies.Contains(p.UpmName)))
            {
                DrawPackageRow(package, state);
            }
            if (group == PackageGroup.ThirdParty) DrawExtraPackages(state);
            List<PackageDefinition> selectedPackages = PackageCatalog.Where(p => p.Group == group && state.IsSelected(p.Id) && !p.IsInstalled).ToList();
            if (group == PackageGroup.UnityRegistry) selectedPackages.AddRange(FeatureBundles.Where(f => state.IsSelected(f.Id) && !IsUpmInstalled(f.Name)).Select(f => f.AsPackageDefinition()));
            EditorGUILayout.BeginHorizontal();
            using (new EditorGUI.DisabledScope(Busy || selectedPackages.Count == 0))
            {
                if (GUILayout.Button("Refresh Selected Versions", GUILayout.Width(195))) RefreshVersions(selectedPackages);
                if (GUILayout.Button("Import Selected", GUILayout.Width(145))) ImportSelected(selectedPackages);
            }
            GUILayout.FlexibleSpace();
            if (GUILayout.Button("Sync Define Symbols", GUILayout.Width(155))) { QueueDefineSync(); operationStatus = "Queued ACS compile-symbol synchronization."; }
            EditorGUILayout.EndHorizontal();
        }

        private void DrawFeatureBundles()
        {
            Box("Unity feature bundles", () =>
            {
                EditorGUILayout.LabelField("Feature bundles are discovered from the configured Unity registries. Their child package rows are hidden to avoid duplicate installation choices.", EditorStyles.wordWrappedMiniLabel);
                if (FeatureBundles.Count == 0 && featureBundleRequest == null && !Busy) RefreshFeatureBundles();
                if (FeatureBundles.Count == 0) EditorGUILayout.LabelField(featureBundleRequest == null ? "No feature bundles were found." : "Loading feature bundles...", mutedStyle);
                foreach (FeatureBundle bundle in FeatureBundles)
                {
                    EditorGUILayout.BeginVertical(EditorStyles.helpBox);
                    EditorGUILayout.BeginHorizontal();
                    bool installed = IsUpmInstalled(bundle.Name);
                    TemplateSetupState state = TemplateSetupState.instance;
                    bool selected = installed || state.IsSelected(bundle.Id);
                    using (new EditorGUI.DisabledScope(installed || Busy))
                    {
                        bool changed = EditorGUILayout.Toggle(selected, GUILayout.Width(18));
                        if (changed != selected) { state.SetSelected(bundle.Id, changed); if (changed) RefreshVersions(new[] { bundle.AsPackageDefinition() }); }
                    }
                    EditorGUILayout.LabelField(bundle.Title, EditorStyles.boldLabel); GUILayout.FlexibleSpace(); GUILayout.Label(installed ? "Ready" : "Not Installed", installed ? readyStyle : warningStyle);
                    EditorGUILayout.EndHorizontal();
                    bundle.Expanded = EditorGUILayout.Foldout(bundle.Expanded, "Included packages (" + bundle.Dependencies.Count + ")", true);
                    if (bundle.Expanded) foreach (string dependency in bundle.Dependencies) EditorGUILayout.LabelField(dependency, mutedStyle);
                    EditorGUILayout.BeginHorizontal();
                    EditorGUILayout.LabelField("Latest: " + LatestLabel(bundle.AsPackageDefinition(), selected), EditorStyles.miniLabel); GUILayout.FlexibleSpace();
                    using (new EditorGUI.DisabledScope(Busy || installed)) if (GUILayout.Button(installed ? "Installed" : "Install", GUILayout.Width(80))) ImportSelected(new[] { bundle.AsPackageDefinition() });
                    using (new EditorGUI.DisabledScope(Busy || !installed)) if (GUILayout.Button("Remove", GUILayout.Width(75))) RemovePackage(bundle.AsPackageDefinition(), state);
                    EditorGUILayout.EndHorizontal(); EditorGUILayout.EndVertical();
                }
                if (GUILayout.Button("Refresh Registry Catalog", GUILayout.Width(180))) RefreshFeatureBundles();
            });
        }

        private void DrawPackageRow(PackageDefinition package, TemplateSetupState state)
        {
            EditorGUILayout.BeginVertical(EditorStyles.helpBox);
            bool installed = package.IsInstalled;
            bool selected = installed || state.IsSelected(package.Id);
            EditorGUILayout.BeginHorizontal();
            using (new EditorGUI.DisabledScope(installed || Busy))
            {
                bool changed = EditorGUILayout.Toggle(selected, GUILayout.Width(18));
                if (changed != selected) { state.SetSelected(package.Id, changed); if (changed) RefreshVersions(new[] { package }); }
            }
            EditorGUILayout.LabelField(package.Title, EditorStyles.boldLabel); GUILayout.FlexibleSpace(); GUILayout.Label(installed ? "Ready" : "Not Installed", installed ? readyStyle : warningStyle);
            EditorGUILayout.EndHorizontal();
            EditorGUILayout.LabelField(package.Description, EditorStyles.wordWrappedMiniLabel);
            EditorGUILayout.BeginHorizontal();
            EditorGUILayout.LabelField("Latest: " + LatestLabel(package, selected), EditorStyles.miniLabel);
            if (installed) EditorGUILayout.LabelField("Installed: " + package.InstalledVersion, EditorStyles.miniLabel);
            EditorGUILayout.EndHorizontal();
            EditorGUILayout.BeginHorizontal();
            if (selected && package.CustomVersionSupported && !installed)
            {
                bool custom = state.IsCustom(package.Id);
                bool changed = EditorGUILayout.ToggleLeft("Custom version", custom, GUILayout.Width(115));
                if (changed != custom) state.SetCustom(package.Id, changed);
                if (state.IsCustom(package.Id))
                {
                    string version = EditorGUILayout.TextField(state.Version(package.Id));
                    if (version != state.Version(package.Id)) state.SetVersion(package.Id, version);
                }
            }
            else if (selected && !package.CustomVersionSupported) GUILayout.Label("Registry and built-in versions are managed by Unity.", mutedStyle);
            GUILayout.FlexibleSpace();
            using (new EditorGUI.DisabledScope(Busy || installed || package.Source == PackageSource.BuiltIn)) if (GUILayout.Button(installed ? "Installed" : "Install", GUILayout.Width(75))) ImportSelected(new[] { package });
            using (new EditorGUI.DisabledScope(Busy || !installed || !package.CanRemove(state))) if (GUILayout.Button("Remove", GUILayout.Width(75))) RemovePackage(package, state);
            EditorGUILayout.EndHorizontal();
            if (installed && !package.CanRemove(state) && package.Source != PackageSource.BuiltIn) EditorGUILayout.LabelField("Existing provider files are unmanaged and protected from removal here.", mutedStyle);
            EditorGUILayout.EndVertical(); EditorGUILayout.Space(4);
        }

        private void DrawExtraPackages(TemplateSetupState state)
        {
            Box("Additional Packages", () =>
            {
                EditorGUILayout.LabelField("Add UPM packages, Git URLs, direct .unitypackage URLs, or purchase links for reference.", EditorStyles.wordWrappedMiniLabel);
                for (int i = 0; i < state.Extras.Count; i++)
                {
                    ExtraPackage extra = state.Extras[i];

                    extra.Name = EditorGUILayout.TextField(extra.Name, GUILayout.Width(135));
                    extra.Kind = (ExtraKind)EditorGUILayout.EnumPopup(extra.Kind, GUILayout.Width(110));
                    extra.Reference = EditorGUILayout.TextField(extra.Reference);

                    EditorGUILayout.EndHorizontal();
                    EditorGUILayout.BeginHorizontal();
                    extra.CustomVersion = EditorGUILayout.ToggleLeft("Custom version", extra.CustomVersion, GUILayout.Width(115));
                    if (extra.CustomVersion)
                    {
                        string version = EditorGUILayout.TextField(extra.Version);
                        if (version != extra.Version) { extra.Version = version; state.SaveState(); }
                    }
                    GUILayout.FlexibleSpace();
                    using (new EditorGUI.DisabledScope(Busy || extra.Kind == ExtraKind.PurchaseLink || string.IsNullOrWhiteSpace(extra.Reference)))
                        if (GUILayout.Button("Import", GUILayout.Width(75))) ImportExtra(extra);
                    EditorGUILayout.EndHorizontal();
                }
                if (GUILayout.Button("Add Package", GUILayout.Width(115))) { state.Extras.Add(new ExtraPackage { Name = "Extra Package", Kind = ExtraKind.Upm }); state.SaveState(); }
            });
        }

        private void DrawPresetRow(PresetProfile profile, TemplateSetupState state, Action properties)
        {
            Preset preset = profile.Load();
            if (preset == null) return;
            EditorGUILayout.BeginVertical(EditorStyles.helpBox);
            EditorGUILayout.BeginHorizontal();
            string renamed = EditorGUILayout.TextField(profile.Name, GUILayout.MinWidth(180));
            if (renamed != profile.Name) { string error = AssetDatabase.RenameAsset(AssetDatabase.GetAssetPath(preset), renamed); if (string.IsNullOrEmpty(error)) { profile.Name = renamed; state.SaveState(); } }
            if (GUILayout.Button(profile.ShowProperties ? "Hide Properties" : "Properties", GUILayout.Width(105))) profile.ShowProperties = !profile.ShowProperties;
            if (GUILayout.Button("Ping", GUILayout.Width(55))) { Selection.activeObject = preset; EditorGUIUtility.PingObject(preset); }
            bool isDefault = state.DefaultGuid(profile.TypeKey) == profile.Guid;
            using (new EditorGUI.DisabledScope(isDefault)) if (GUILayout.Button("Set Default", GUILayout.Width(88))) { state.SetDefault(profile.TypeKey, profile.Guid); }
            if (GUILayout.Button("Delete", GUILayout.Width(58))) DeletePreset(profile, state);
            if (isDefault) GUILayout.Label("Default", readyStyle);
            EditorGUILayout.EndHorizontal();
            if (profile.ShowProperties) properties();
            EditorGUILayout.EndVertical();
        }

        private void DeletePreset(PresetProfile profile, TemplateSetupState state)
        {
            Preset preset = profile.Load();
            if (preset == null) return;
            string message = "Delete '" + preset.name + "' from this project? This permanently removes the preset asset.\n\nProject assets are not changed.";
            if (!EditorUtility.DisplayDialog("Delete Template Preset", message, "Delete Preset", "Cancel")) return;
            BeginOperation("Deleting " + preset.name + "...");
            string assetPath = AssetDatabase.GetAssetPath(preset);
            if (Selection.activeObject == preset) Selection.activeObject = null;
            if (!AssetDatabase.DeleteAsset(assetPath)) { EndOperation("Preset deletion failed."); return; }
            state.RemoveProfile(profile);
            EndOperation("Preset deleted from the project.");
            GUIUtility.ExitGUI();
        }

        private void CreateTextureProfile(TemplateSetupState state)
        {
            BeginOperation("Creating TextureImportPreset...");
            const string temporaryPath = "Assets/TemplateSetup_TemporaryTexture.png";
            Texture2D texture = new Texture2D(2, 2);
            try
            {
                texture.SetPixels(new[] { Color.white, Color.white, Color.white, Color.white }); texture.Apply(); File.WriteAllBytes(temporaryPath, texture.EncodeToPNG()); AssetDatabase.ImportAsset(temporaryPath, ImportAssetOptions.ForceSynchronousImport);
                TextureImporter importer = AssetImporter.GetAtPath(temporaryPath) as TextureImporter; CreateFriendlyPreset(new Preset(importer), "TextureImportPreset", p => state.TextureProfiles.Add(TextureImportPresetProfile.FromPreset(p)));
                state.SetDefaultIfEmpty("texture", state.TextureProfiles.Last().Guid);
            }
            finally { DestroyImmediate(texture); AssetDatabase.DeleteAsset(temporaryPath); EndOperation("TextureImportPreset created."); }
        }

        private void CreateImageProfile(TemplateSetupState state)
        {
            BeginOperation("Creating ImagePreset...");
            GameObject source = new GameObject("TemplatePresetSource", typeof(Image));
            try { CreateFriendlyPreset(new Preset(source.GetComponent<Image>()), "ImagePreset", p => state.ImageProfiles.Add(ImagePresetProfile.FromPreset(p))); state.SetDefaultIfEmpty("image", state.ImageProfiles.Last().Guid); }
            finally { DestroyImmediate(source); EndOperation("ImagePreset created."); }
        }

        private void CreateTextProfile(TemplateSetupState state)
        {
            BeginOperation("Creating TextMeshProUGUIPreset...");
            GameObject source = new GameObject("TemplatePresetSource", typeof(TextMeshProUGUI));
            try { CreateFriendlyPreset(new Preset(source.GetComponent<TextMeshProUGUI>()), "TextMeshProUGUIPreset", p => state.TextProfiles.Add(TextMeshProUGUIPresetProfile.FromPreset(p))); state.SetDefaultIfEmpty("text", state.TextProfiles.Last().Guid); }
            finally { DestroyImmediate(source); EndOperation("TextMeshProUGUIPreset created."); }
        }

        private static void CreateFriendlyPreset(Preset preset, string name, Action<Preset> registered)

        private static void RefreshFeatureBundles()
        {
            if (featureBundleRequest != null || Busy && featureBundleRequest == null) return;

        private static void PollFeatureBundles()
        {
            if (featureBundleRequest == null || !featureBundleRequest.IsCompleted) return;
            EditorApplication.update -= PollFeatureBundles; SearchRequest done = featureBundleRequest; featureBundleRequest = null; FeatureBundles.Clear();
            if (done.Status == StatusCode.Success && done.Result != null)
            {
                foreach (UnityEditor.PackageManager.PackageInfo info in done.Result.Where(p => p.name.StartsWith("com.unity.feature.", StringComparison.OrdinalIgnoreCase)).OrderBy(p => p.displayName))

        }
        private static void RefreshVersions(IEnumerable<PackageDefinition> packages)
        {
            if (Busy) return;
            versionQueue = new Queue<PackageDefinition>(packages); NextVersion();
        }
        private static void NextVersion()
        {
            if (versionQueue == null || versionQueue.Count == 0) { versionQueue = null; activePackage = null; operationStatus = "Latest version check complete."; return; }
            activePackage = versionQueue.Dequeue(); operationStatus = "Checking latest version for " + activePackage.Title + "..."; VersionErrors.Remove(activePackage.Id);
            if (activePackage.Source == PackageSource.Upm) { searchRequest = Client.Search(activePackage.UpmName); EditorApplication.update += PollSearch; }
            else StartGithubRequest(activePackage, false, null);
        }
        private static void PollSearch()
        {
            if (searchRequest == null || !searchRequest.IsCompleted) return;
            EditorApplication.update -= PollSearch; SearchRequest done = searchRequest; searchRequest = null;
            if (done.Status == StatusCode.Success && done.Result != null && done.Result.Length > 0) LatestVersions[activePackage.Id] = done.Result[0].versions.latestCompatible ?? done.Result[0].versions.latest ?? done.Result[0].version;
            else VersionErrors[activePackage.Id] = done.Error?.message ?? "Package is unavailable from the configured registries.";
            NextVersion();
        }
        private static void StartGithubRequest(PackageDefinition package, bool installation, string customVersion)
        {
            string url = package.VersionUrl;
            if (!string.IsNullOrWhiteSpace(customVersion))
            {
                string tag = customVersion.StartsWith("v", StringComparison.OrdinalIgnoreCase) ? customVersion : "v" + customVersion;
                url = "https://api.github.com/repos/" + package.GithubRepo + "/releases/tags/" + Uri.EscapeDataString(tag);
            }
            activePackage = package; webRequest = UnityWebRequest.Get(url); webRequest.SetRequestHeader("User-Agent", "UnityTemplateSetup"); webRequest.SetRequestHeader("Accept", "application/vnd.github+json"); webRequest.SendWebRequest(); EditorApplication.update += PollGithub;
            SessionState.SetBool("TemplateSetup.GitHubInstall", installation); SessionState.SetString("TemplateSetup.GitHubVersion", customVersion ?? string.Empty);
        }
        private static void PollGithub()
        {
            if (webRequest == null || !webRequest.isDone) return;
            EditorApplication.update -= PollGithub; UnityWebRequest done = webRequest; webRequest = null;
            bool installation = SessionState.GetBool("TemplateSetup.GitHubInstall", false); string customVersion = SessionState.GetString("TemplateSetup.GitHubVersion", string.Empty);
            if (done.result != UnityWebRequest.Result.Success) { HandleGithubFailure(done.error ?? "GitHub request failed.", installation); done.Dispose(); return; }
            GithubRelease release = JsonUtility.FromJson<GithubRelease>(done.downloadHandler.text); done.Dispose();
            if (release == null || string.IsNullOrWhiteSpace(release.tag_name)) { HandleGithubFailure("GitHub returned no release tag.", installation); return; }
            Releases[activePackage.Id] = release; LatestVersions[activePackage.Id] = release.tag_name;
            if (!installation) { NextVersion(); return; }
            GithubAsset asset = release.assets?.FirstOrDefault(a => a.name != null && a.name.EndsWith(".unitypackage", StringComparison.OrdinalIgnoreCase) && activePackage.AssetWords.Any(w => a.name.IndexOf(w, StringComparison.OrdinalIgnoreCase) >= 0));
            if (asset == null) { FailImport("The selected release does not contain the expected .unitypackage asset."); return; }
            DownloadUnityPackage(asset.browser_download_url);
        }
        private static void HandleGithubFailure(string error, bool installation) { if (installation) FailImport(error); else { VersionErrors[activePackage.Id] = error; NextVersion(); } }

        private static void ImportSelected(IEnumerable<PackageDefinition> packages)
        {
            PackageDefinition[] missing = packages.Where(p => !p.IsInstalled).ToArray();
            if (missing.Length == 0) { operationStatus = "All selected packages are already ready."; return; }
            string[] upm = missing.Where(p => p.Source == PackageSource.Upm).Select(p =>
            {
                string custom = TemplateSetupState.instance.IsCustom(p.Id) ? TemplateSetupState.instance.Version(p.Id) : string.Empty;
                return string.IsNullOrWhiteSpace(custom) ? p.UpmName : p.UpmName + "@" + custom.Trim();
            }).ToArray();
            importQueue = new Queue<PackageDefinition>(missing.Where(p => p.Source != PackageSource.Upm));
            if (upm.Length == 0) { NextExternalImport(); return; }
            operationStatus = "Resolving selected UPM packages..."; packageRequest = Client.AddAndRemove(upm, Array.Empty<string>()); EditorApplication.update += PollUpmImport;
        }
        private static void PollUpmImport()
        {
            if (packageRequest == null || !packageRequest.IsCompleted) return;
            EditorApplication.update -= PollUpmImport; AddAndRemoveRequest done = packageRequest; packageRequest = null;
            if (done.Status != StatusCode.Success) { operationStatus = "Package installation failed: " + done.Error?.message; importQueue = null; return; }
            RefreshInstalledPackageCache();
            NextExternalImport();
        }
        private static void NextExternalImport()
        {
            if (importQueue == null || importQueue.Count == 0) { importQueue = null; activePackage = null; QueueDefineSync(); operationStatus = "Selected packages imported. ACS compile-symbol synchronization was queued."; return; }
            activePackage = importQueue.Dequeue(); operationStatus = "Downloading " + activePackage.Title + "...";
            if (activePackage.Source == PackageSource.AppLovinInstaller) { DownloadUnityPackage(AppLovinInstallerUrl); return; }
            string custom = TemplateSetupState.instance.IsCustom(activePackage.Id) ? TemplateSetupState.instance.Version(activePackage.Id) : null;
            StartGithubRequest(activePackage, true, custom);
        }
        private static void DownloadUnityPackage(string url)
        {
            string folder = Path.Combine(Path.GetTempPath(), "TemplateSetupDownloads"); Directory.CreateDirectory(folder);
            downloadPath = Path.Combine(folder, activePackage.Id + "-" + DateTime.UtcNow.Ticks + ".unitypackage");
            webRequest = UnityWebRequest.Get(url); webRequest.downloadHandler = new DownloadHandlerFile(downloadPath); webRequest.SendWebRequest(); EditorApplication.update += PollDownload;
        }
        private static void PollDownload()
        {
            if (webRequest == null || !webRequest.isDone) return;
            EditorApplication.update -= PollDownload; UnityWebRequest done = webRequest; webRequest = null;
            if (done.result != UnityWebRequest.Result.Success) { FailImport(done.error ?? "Download failed."); done.Dispose(); return; }
            done.Dispose(); importingUnityPackage = true; operationStatus = "Importing " + activePackage.Title + "..."; AssetDatabase.ImportPackage(downloadPath, false);
        }

            if (!importingUnityPackage) return;
            importingUnityPackage = false; TemplateSetupState.instance.MarkManaged(activePackage.Id); NextExternalImport();

        private static void RemovePackage(PackageDefinition package, TemplateSetupState state)

        }

        {
            if (packageRequest == null || !packageRequest.IsCompleted) return;

"Package removed." : "Package removal failed: " + done.Error?.message; if (done.Status == StatusCode.Success) { RefreshInstalledPackageCache(); QueueDefineSync(); }

        private static string LatestLabel(PackageDefinition package, bool selected)
        {

            return VersionErrors.TryGetValue(package.Id, out string error) ? "unavailable (" + error + ")" : "checking...";
        }

        [FilePath("ProjectSettings/TemplateSetupState.asset", FilePathAttribute.Location.ProjectFolder)]
        private sealed class TemplateSetupState : ScriptableSingleton<TemplateSetupState>
        {
            [SerializeField] private List<string> selected = new(); [SerializeField] private List<string> custom = new(); [SerializeField] private List<string> versions = new(); [SerializeField] private List<string> managed = new(); [SerializeField] private List<ExtraPackage> extras = new();
            [SerializeField] private List<TextureImportPresetProfile> textureProfiles = new(); [SerializeField] private List<ImagePresetProfile> imageProfiles = new(); [SerializeField] private List<TextMeshProUGUIPresetProfile> textProfiles = new();
            [SerializeField] private string defaultTextureGuid; [SerializeField] private string defaultImageGuid; [SerializeField] private string defaultTextGuid;
            public List<ExtraPackage> Extras => extras;
            public List<TextureImportPresetProfile> TextureProfiles => textureProfiles; public List<ImagePresetProfile> ImageProfiles => imageProfiles; public List<TextMeshProUGUIPresetProfile> TextProfiles => textProfiles;
            public bool IsSelected(string id) => selected.Contains(id); public void SetSelected(string id, bool value) { if (value) { if (!selected.Contains(id)) selected.Add(id); } else selected.Remove(id); SaveState(); }
            public bool IsCustom(string id) => custom.Contains(id); public void SetCustom(string id, bool value) { if (value) { if (!custom.Contains(id)) { custom.Add(id); versions.Add(string.Empty); } } else { int i = custom.IndexOf(id); if (i >= 0) { custom.RemoveAt(i); if (i < versions.Count) versions.RemoveAt(i); } } SaveState(); }
            public string Version(string id) { int i = custom.IndexOf(id); return i >= 0 && i < versions.Count ? versions[i] : string.Empty; } public void SetVersion(string id, string value) { int i = custom.IndexOf(id); if (i < 0) { custom.Add(id); versions.Add(value); } else { while (versions.Count <= i) versions.Add(string.Empty); versions[i] = value; } SaveState(); }
            public bool IsManaged(string id) => managed.Contains(id); public void MarkManaged(string id) { if (!managed.Contains(id)) managed.Add(id); SaveState(); } public void UnmarkManaged(string id) { managed.Remove(id); SaveState(); } public void SaveState() => Save(true);
            public string DefaultGuid(string type) => type == "texture" ? defaultTextureGuid : type == "image" ? defaultImageGuid : defaultTextGuid;
            public void SetDefault(string type, string guid) { if (type == "texture") defaultTextureGuid = guid; else if (type == "image") defaultImageGuid = guid; else defaultTextGuid = guid; SaveState(); }
            public void SetDefaultIfEmpty(string type, string guid) { if (string.IsNullOrEmpty(DefaultGuid(type))) SetDefault(type, guid); }
            public void RemoveProfile(PresetProfile profile)
            {
                if (profile.TypeKey == "texture") textureProfiles.RemoveAll(p => p.Guid == profile.Guid);
                else if (profile.TypeKey == "image") imageProfiles.RemoveAll(p => p.Guid == profile.Guid);
                else textProfiles.RemoveAll(p => p.Guid == profile.Guid);
                if (DefaultGuid(profile.TypeKey) == profile.Guid)
                {
                    string replacement = profile.TypeKey == "texture" ? textureProfiles.FirstOrDefault()?.Guid : profile.TypeKey == "image" ? imageProfiles.FirstOrDefault()?.Guid : textProfiles.FirstOrDefault()?.Guid;
                    if (profile.TypeKey == "texture") defaultTextureGuid = replacement; else if (profile.TypeKey == "image") defaultImageGuid = replacement; else defaultTextGuid = replacement;
                }
                SaveState();
            }
            public void SyncProfiles(IEnumerable<Preset> presets)
            {
                foreach (Preset preset in presets)
                {
                    string guid = AssetDatabase.AssetPathToGUID(AssetDatabase.GetAssetPath(preset)); if (string.IsNullOrEmpty(guid)) continue;
                    string target = preset.GetTargetTypeName();
                    if (target == "TextureImporter" && textureProfiles.All(p => p.Guid != guid)) textureProfiles.Add(TextureImportPresetProfile.FromPreset(preset));
                    else if (target == typeof(Image).FullName && imageProfiles.All(p => p.Guid != guid)) imageProfiles.Add(ImagePresetProfile.FromPreset(preset));
                    else if (target == typeof(TextMeshProUGUI).FullName && textProfiles.All(p => p.Guid != guid)) textProfiles.Add(TextMeshProUGUIPresetProfile.FromPreset(preset));
                }
                textureProfiles.RemoveAll(p => p.Load() == null); imageProfiles.RemoveAll(p => p.Load() == null); textProfiles.RemoveAll(p => p.Load() == null);
                if (string.IsNullOrEmpty(defaultTextureGuid) && textureProfiles.Count > 0) defaultTextureGuid = textureProfiles[0].Guid;
                if (string.IsNullOrEmpty(defaultImageGuid) && imageProfiles.Count > 0) defaultImageGuid = imageProfiles[0].Guid;
                if (string.IsNullOrEmpty(defaultTextGuid) && textProfiles.Count > 0) defaultTextGuid = textProfiles[0].Guid;
                SaveState();
            }
        }

            {
